## 1. 迷你 Redux（createStore）

**考察点一句话：单一状态 + 纯函数 reducer + 发布订阅。**

```js
function createStore(reducer, preloadedState) {
  let state = preloadedState;
  const listeners = [];

  const getState = () => state;

  const dispatch = (action) => {
    state = reducer(state, action);      // 纯函数算新状态（不可变更新）
    listeners.forEach(l => l());         // 通知订阅者
    return action;
  };

  const subscribe = (listener) => {
    listeners.push(listener);
    return () => {                        // 返回取消订阅函数
      const i = listeners.indexOf(listener);
      if (i > -1) listeners.splice(i, 1);
    };
  };

  dispatch({ type: '@@INIT' });          // 初始化：让 reducer 走 default 返回初始 state
  return { getState, dispatch, subscribe };
}
```

**进阶：applyMiddleware + thunk（洋葱模型）**

```js
const compose = (...fns) => fns.reduce((a, b) => (...args) => a(b(...args)));

function applyMiddleware(...middlewares) {
  return (createStore) => (reducer, preloadedState) => {
    const store = createStore(reducer, preloadedState);
    let dispatch = () => { throw new Error('dispatch 初始化中'); };
    const middlewareAPI = {
      getState: store.getState,
      dispatch: (action, ...args) => dispatch(action, ...args),
    };
    const chain = middlewares.map(m => m(middlewareAPI));
    dispatch = compose(...chain)(store.dispatch);   // 层层包装 dispatch
    return { ...store, dispatch };
  };
}

// thunk 中间件：让 dispatch 能接收函数（异步逻辑的核心）
const thunk = ({ dispatch, getState }) => next => action => {
  if (typeof action === 'function') {
    return action(dispatch, getState);
  }
  return next(action);
};
```

## 2. 虚拟列表

**考察点一句话：空容器撑高度 + scrollTop 算范围 + translateY 定位（只渲染可视区）。**

```jsx
function VirtualList({ items, itemHeight, height }) {
  const [scrollTop, setScrollTop] = useState(0);

  const startIndex = Math.floor(scrollTop / itemHeight);        // 可视区第一条
  const visibleCount = Math.ceil(height / itemHeight) + 5;      // +5 = overscan 缓冲
  const endIndex = startIndex + visibleCount;
  const visibleItems = items.slice(startIndex, endIndex);

  return (
    <div
      style={{ height, overflowY: 'auto' }}
      onScroll={e => setScrollTop(e.target.scrollTop)}
    >
      {/* 空容器撑出滚动条：总高度 = 数据量 × 行高 */}
      <div style={{ height: items.length * itemHeight, position: 'relative' }}>
        {visibleItems.map((item, i) => (
          <div
            key={startIndex + i}
            style={{
              position: 'absolute',
              top: 0,
              transform: `translateY(${(startIndex + i) * itemHeight}px)`,  // 定位
              height: itemHeight,
            }}
          >
            {item}
          </div>
        ))}
      </div>
    </div>
  );
}
```

**进阶：动态行高 = 预估 + 测量 + 缓存（说出思路即可，不用写全）。**
滚动优化：onScroll 用 requestAnimationFrame 节流 + `{ passive: true }`。

## 3. 迷你 useState（体现 Hook 原理）

**考察点一句话：状态存数组 + 游标按调用顺序对应（这就是为什么不能条件调用）。**

```js
let state = [];      // 对应 fiber.memoizedState 链表
let cursor = 0;      // 游标：第 N 次调用读数组第 N 个位置

function useState(initial) {
  const currentCursor = cursor;
  if (state[currentCursor] === undefined) {
    state[currentCursor] = initial;
  }
  const setState = (newVal) => {
    state[currentCursor] = newVal;
    render();        // 触发重渲染（真实 React：排队 + 调度，不立即执行）
  };
  cursor++;
  return [state[currentCursor], setState];
}

function render() {
  cursor = 0;        // 每次渲染重置游标 → 靠调用顺序一一对应
  // ReactDOM.render(<App />, root) 或 React 18 的 createRoot().render()
}
```

**口头解释（必背）：** "useState 的状态存在数组/链表里，靠**调用顺序**（游标）对应。如果在 if 条件里调用，某次渲染少调一次，游标就错位，后续所有状态都拿错——所以 Hook 必须在顶层调用。"

## 4. 迷你 useEffect（deps + cleanup）

**考察点一句话：deps 变化才执行，执行前先跑上次的 cleanup（先 destroy 后 create）。**

```js
const effects = [];
let cursor = 0;

function useEffect(cb, deps) {
  const currentCursor = cursor;
  const prev = effects[currentCursor];
  const changed = !prev || !deps ||
    deps.some((d, i) => d !== prev.deps[i]);       // Object.is 逐个比较

  if (changed) {
    if (prev?.cleanup) prev.cleanup();             // 先清理旧的
    const cleanup = cb();                          // 再执行新的
    effects[currentCursor] = { deps, cleanup };
  }
  cursor++;
}
```

**口头解释：** "deps 不传 = 每次渲染都执行；传 [] = 只在 mount 执行一次（prev 不存在）；cleanup 在下次执行前先跑，组件卸载时 React 也会调 cleanup。"

## 5. 迷你 useMemo / useCallback

**考察点一句话：useCallback 就是 useMemo(() => fn, deps) 的特例。**

```js
const memoCache = [];
let cursor = 0;

function useMemo(fn, deps) {
  const currentCursor = cursor;
  const prev = memoCache[currentCursor];
  const changed = !prev || deps.some((d, i) => d !== prev.deps[i]);
  if (changed) {
    memoCache[currentCursor] = { deps, value: fn() };   // 只有 deps 变了才重算
  }
  cursor++;
  return memoCache[currentCursor].value;
}

function useCallback(fn, deps) {
  return useMemo(() => fn, deps);   // 缓存函数本身（引用稳定）
}
```

**口头解释：** "两者都是 deps 驱动的缓存：useMemo 缓存**返回值**，useCallback 缓存**函数引用**。deps 不变就返回上次缓存，变了才重新计算——这是引用稳定性的原理。"

## 6. 迷你调度器（时间切片）

**考察点一句话：MessageChannel 宏任务 + 5ms 时间片 + 让出主线程。**

```js
const taskQueue = [];
let isFlushing = false;

function scheduleCallback(task) {
  taskQueue.push(task);
  if (!isFlushing) {
    isFlushing = true;
    // MessageChannel：宏任务，无 setTimeout 的 4ms clamp，时机可控
    const channel = new MessageChannel();
    channel.port1.onmessage = () => {
      isFlushing = false;
      flushWork();
    };
    channel.port2.postMessage(null);
  }
}

function flushWork() {
  const startTime = performance.now();
  while (taskQueue.length) {
    if (performance.now() - startTime >= 5) break;   // 5ms 时间片到期 → 让出主线程
    taskQueue.shift()();                              // 执行一个任务
  }
  if (taskQueue.length) scheduleCallback(flushWork);  // 没干完，下个宏任务继续
}
```

**口头解释：** "调度器把任务拆成 5ms 一片，每片执行完通过 MessageChannel 排一个宏任务让出主线程，浏览器趁机响应输入和绘制；下次宏任务从断点继续。这就是时间切片的最小实现。"

const handlers = {};

## 8. 手写 useSyncExternalStore（Zustand 的原理基础）

**考察点一句话：订阅外部 store + 变化时 setState 触发更新（selector 细粒度是它的进阶）。**

```js
function useSyncExternalStore(subscribe, getSnapshot) {
  const [state, setState] = useState(getSnapshot);     // 初始快照

  useEffect(() => {
    // 订阅外部 store，变化时用新快照触发重渲染
    return subscribe(() => {
      const next = getSnapshot();
      setState(next);
    });
  }, [subscribe, getSnapshot]);

  return state;
}

// 使用（Zustand 的简化版用法）
function useStore(selector) {
  return useSyncExternalStore(
    store.subscribe,
    () => selector(store.getState()),   // 只订阅选中的 slice
  );
}
```

**口头解释：** "useSyncExternalStore 是 React 18 提供的订阅外部 store 的标准 API：subscribe 注册监听，getSnapshot 提供快照。Zustand 就是基于它 + selector 输出比较实现细粒度渲染的——selector 结果没变就不重渲染。"

## 36. 简版 useState（闭包 + 链表思想）

**考察点一句话：hook 存链表（对应 fiber.memoizedState）+ 闭包捕获节点 + 按顺序复用（条件调用会错位）。**

```js
function createComponent(renderFn) {
  let hooks = null;        // 链表头（对应 fiber.memoizedState）
  let currentHook = null;  // 当前遍历指针
  let isMount = true;      // 是否首次挂载

  function useState(initial) {
    let hook;
    if (isMount) {
      // 挂载：创建链表节点
      hook = { state: initial, next: null };
      if (!hooks) { hooks = hook; currentHook = hook; }
      else { currentHook.next = hook; currentHook = hook; }
    } else {
      // 更新：按调用顺序复用链表节点（不重建！）
      hook = currentHook;
      currentHook = currentHook.next;
    }

    // 闭包捕获 hook 节点 → setState 永远操作同一个节点
    const setState = (newVal) => {
      hook.state = typeof newVal === 'function' ? newVal(hook.state) : newVal;
      rerender();   // 真实 React：排队 + 调度（不立即执行）
    };

    return [hook.state, setState];
  }

  function rerender() {
    currentHook = hooks;   // 回到链表头，从头按顺序遍历
    renderFn();
  }

  return { useState, rerender };
}
```

**口头解释：** "状态存在链表的节点上，setState 通过**闭包**捕获住自己对应的那个节点，所以引用恒定、永远改对地方。每次渲染从链表头按**调用顺序**遍历——如果组件里条件调用了 useState，链表遍历顺序就错位，后面的 hook 全拿错状态，这就是 Hook 必须顶层调用的原因。"

## 37. 简版 useEffect（依赖对比）

**考察点一句话：deps 用 Object.is 逐个比较 + 先 destroy 后 create。**

```js
// 在 createComponent 里扩展（hooks 是同一个链表）
function useEffect(callback, deps) {
  let hook;
  if (isMount) {
    hook = { state: { deps: undefined, cleanup: undefined }, next: null };
    if (!hooks) { hooks = hook; currentHook = hook; }
    else { currentHook.next = hook; currentHook = hook; }
  } else {
    hook = currentHook;
    currentHook = currentHook.next;
  }

  const prevDeps = hook.state.deps;
  const changed = !prevDeps ||              // 首次执行
    !deps ||                                // 不传 deps → 每次渲染都执行
    deps.some((d, i) => d !== prevDeps[i]); // Object.is 逐个比较

  if (changed) {
    if (hook.state.cleanup) hook.state.cleanup();  // ① 先销毁旧的
    const cleanup = callback();                     // ② 再执行新的
    hook.state = { deps, cleanup };
  }
}
```

**口头解释：** "deps 对比是浅比较（Object.is 逐个元素），不是深比较。执行顺序永远是**先 destroy 旧的、再 create 新的**——这是 React 保证副作用不泄漏的关键。组件卸载时 React 会额外调用所有 cleanup。"

## 38. useFetch（取消请求 + loading + error）

**考察点一句话：AbortController 中断请求 + isCancelled 防竞态 + cleanup 里清理。**

```js
function useFetch(url, options = {}) {
  const [data, setData] = useState(null);
  const [loading, setLoading] = useState(true);
  const [error, setError] = useState(null);

  useEffect(() => {
    const controller = new AbortController();  // ① 取消请求的控制器
    let isCancelled = false;                   // ② 防竞态标记

    const fetchData = async () => {
      setLoading(true);
      setError(null);
      try {
        const res = await fetch(url, { ...options, signal: controller.signal });
        if (!res.ok) throw new Error(`HTTP ${res.status}`);
        const json = await res.json();
        if (!isCancelled) setData(json);       // ③ 已卸载就不 setState
      } catch (e) {
        if (e.name === 'AbortError') return;   // ④ 主动取消的报错忽略
        if (!isCancelled) setError(e.message);
      } finally {
        if (!isCancelled) setLoading(false);
      }
    };

    fetchData();

    // ⑤ 清理：组件卸载或 url 变化时
    return () => {
      isCancelled = true;     // 标记：旧请求回来不再 setState（防竞态）
      controller.abort();     // 真正中断网络请求
    };
  }, [url]);   // 依赖 url，变化就重新请求

  return { data, loading, error };
}

// 使用
const { data, loading, error } = useFetch('/api/user/123');
```

**口头解释（为什么防竞态）：** "用户快速切换 url 时，旧请求可能比新请求后返回（竞态）。isCancelled 标记保证**只有最后一次请求的结果会被 setState**——这是 useFetch 最容易漏的点，面试主动讲出来很加分。"

## 39. 简版 Virtual DOM 和 Diff 算法

**考察点一句话：h 创建 VNode + 递归创建 DOM + 同层 diff（类型不同替换、相同递归 children）。**

```js
// ① 创建 VNode（类似 createElement）
function h(type, props = {}, ...children) {
  return { type, props, children: children.flat() };
}

// ② VNode → 真实 DOM
function createElement(vnode) {
  if (typeof vnode === 'string') return document.createTextNode(vnode);
  const el = document.createElement(vnode.type);
  Object.entries(vnode.props || {}).forEach(([k, v]) => {
    if (k.startsWith('on')) el.addEventListener(k.slice(2).toLowerCase(), v);
    else el.setAttribute(k, v);
  });
  vnode.children.forEach(child => el.appendChild(createElement(child)));
  return el;
}

// ③ 简易 Diff：同层比较（对应 React 的"只比较同层"假设）
function diff(parent, oldNode, newNode, index = 0) {
  const oldEl = parent.childNodes[index];

  if (!oldNode && newNode) {
    parent.appendChild(createElement(newNode));      // 新增
  } else if (oldNode && !newNode) {
    parent.removeChild(oldEl);                        // 删除
  } else if (typeof oldNode === 'string' || typeof newNode === 'string') {
    if (oldNode !== newNode) oldEl.textContent = newNode;  // 文本变化
  } else if (oldNode.type !== newNode.type) {
    parent.replaceChild(createElement(newNode), oldEl);    // 类型不同 → 整体替换
  } else {
    updateProps(oldEl, oldNode.props, newNode.props);     // 类型相同 → 更新属性
    const maxLen = Math.max(oldNode.children.length, newNode.children.length);
    for (let i = 0; i < maxLen; i++) {
      diff(oldEl, oldNode.children[i], newNode.children[i], i);  // 递归同层 children
    }
  }
}
```

**口头解释（必背）：** "这个实现体现了 React diff 的核心假设：**只比较同层**（递归时固定 index）、**类型不同直接替换**（不深入比较子树）、**文本节点单独处理**。真实 React 多了 key 复用和移动判断（lastPlacedIndex），但骨架就是这个：同层递归 + 类型判断 + 属性更新。"

## 40. React 简版事件委托机制

**考察点一句话：委托到根容器统一监听 + 合成事件对象 + 模拟捕获/冒泡两阶段。**

```js
class SimpleReact {
  constructor(root) {
    this.root = root;
    // ① 委托：根容器统一监听（React 委托到 root，不是每个元素）
    ['click', 'input', 'change'].forEach(type => {
      root.addEventListener(type, e => this.dispatch(e), false);
    });
  }

  // ② 合成事件对象：包装原生事件，统一接口
  wrapEvent(e) {
    const synthetic = {
      nativeEvent: e,
      target: e.target,
      preventDefault: () => e.preventDefault(),
      stopPropagation: () => { e.stopPropagation(); synthetic._stopped = true; },
    };
    return synthetic;
  }

  dispatch(e) {
    const synthetic = this.wrapEvent(e);
    // ③ 收集冒泡路径（对应 React 沿 fiber 树向上收集）
    const path = [];
    let node = e.target;
    while (node && node !== this.root) {
      path.push(node);
      node = node.parentNode;
    }

    // ④ 捕获阶段：根 → 目标
    for (let i = path.length - 1; i >= 0; i--) {
      const el = path[i];
      const fn = el.__handlers?.[e.type]?.capture;
      if (fn) { fn.call(el, synthetic); if (synthetic._stopped) break; }
    }
    // ⑤ 冒泡阶段：目标 → 根
    for (let i = 0; i < path.length; i++) {
      const el = path[i];
      const fn = el.__handlers?.[e.type]?.bubble;
      if (fn) { fn.call(el, synthetic); if (synthetic._stopped) break; }
    }
  }
}

// 绑定：不直接 addEventListener，而是存到元素上，由委托统一派发
function attachHandler(el, type, fn, phase = 'bubble') {
  el.__handlers = el.__handlers || {};
  el.__handlers[type] = el.__handlers[type] || {};
  el.__handlers[type][phase] = fn;
}

// 使用：动态添加的元素天然生效（委托的好处）
const app = new SimpleReact(document.getElementById('root'));
attachHandler(btn, 'click', (e) => console.log('点击了', e.target));
```

**口头解释：** "委托 = 事件统一挂根容器（内存少、动态元素免绑定）；合成对象 = 统一接口（target/preventDefault/stopPropagation）；两阶段 = 先捕获（根到目标）再冒泡（目标到根）。真实 React 的区别：收集路径沿 **fiber 树**而不是 DOM 树，所以合成事件的 stopPropagation 拦不住原生冒泡（原生早已冒到根）。"
